cmake_minimum_required(VERSION 3.16)
project(forge_web VERSION 0.1.4 LANGUAGES C)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WEB REQUIRED IMPORTED_TARGET libmicrohttpd json-c libcurl openssl)
find_package(Threads REQUIRED)
add_library(forge_web STATIC src/bridge.c)
target_include_directories(forge_web PUBLIC include)
target_link_libraries(forge_web PUBLIC PkgConfig::WEB Threads::Threads)
target_compile_features(forge_web PUBLIC c_std_11)
target_compile_options(forge_web PRIVATE -Wall -Wextra -Werror)
include(CTest)
if(BUILD_TESTING)
    add_executable(bridge_test tests/bridge_test.c)
    target_link_libraries(bridge_test PRIVATE forge_web)
    target_compile_options(bridge_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME bridge_scopes_and_validation COMMAND bridge_test)
endif()
if(BUILD_TESTING)
    add_executable(native_limits_test tests/native_limits_test.c)
    target_include_directories(native_limits_test PRIVATE include)
    target_link_libraries(native_limits_test PRIVATE PkgConfig::WEB Threads::Threads)
    target_compile_options(native_limits_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
    add_test(NAME native_limits_concurrency_and_cleanup COMMAND native_limits_test)
    add_executable(http_client_test tests/http_client_test.c)
    target_link_libraries(http_client_test PRIVATE forge_web)
    target_compile_options(http_client_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()

    target_link_libraries(http_server_test PRIVATE forge_web)
    target_compile_options(http_server_test PRIVATE -UNDEBUG -Wall -Wextra -Werror)
endif()

    add_executable(security_test tests/security_test.c)
    target_include_directories(security_test PRIVATE include)
